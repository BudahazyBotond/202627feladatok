class ImageController extends Controller
{
    public function index(Request $request):JsonResponse {
    $queryBuilder = DB::table("images");
    //$queryBuilder->where("tag","=","latest");
    //$queryBuilder->where("repository","like","docs/%");
    if( $request->filled("orderBy") ) {
        $queryBuilder->orderBy($request->query("orderBy"));
    }
    return response()->json([
        "data"=> $queryBuilder->get(),
    ]);
    }

    public function show(int $id):JsonResponse {
        $queryBuilder = DB::table("images")
            ->where("id", $id);

        return response()->json([
            "data"=> $queryBuilder->firstOrFail(),
        ]);
    }
}

`api.php`
Route::get("/images",[ImageController::class,"index"])->name("images.name");
Route::get("/images/{id}",[ImageController::class,"show"])->name("images.show");
